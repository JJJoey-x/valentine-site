<script>

let stage = 0;

let yesCount = 0;

let noCount = 0;

const message = document.getElementById('message');

const envelope = document.getElementById('envelope');

const letter = document.getElementById('letter');

const buttons = document.getElementById('buttons');

const cat = document.getElementById('cat');

const finger = document.getElementById('finger');

const music = document.getElementById('music');

const rainSound = document.getElementById('rainSound');

const popSound = document.getElementById('popSound');


// Autoplay music after first click

document.body.addEventListener('click', ()=>{ music.play().catch(()=>{}); },{once:true});


// Hearts + sparkles

setInterval(()=>{

  const heart = document.createElement('div');

  heart.className='heart';

  heart.innerText='❤️';

  heart.style.left=Math.random()*100+'vw';

  document.body.appendChild(heart);

  heart.onclick=()=>{ popSound.play(); heart.remove(); };

  setTimeout(()=>heart.remove(),5000);

},400);


setInterval(()=>{

  const sp = document.createElement('div');

  sp.className='sparkle';

  sp.innerText='✨';

  sp.style.left=Math.random()*100+'vw';

  document.body.appendChild(sp);

  setTimeout(()=>sp.remove(),3000);

},800);


setTimeout(()=>{message.innerText='Please 🥺';},1000);


envelope.onclick = ()=>{

  envelope.classList.add('open');

  letter.style.display='block';

  letter.innerText="You are the best thing that's ever happened to me. I want to be with you for time to come and I can't imagine losing you or being without you. I love you for you and everything about you. I love that you love me for me. I thank you for being the most loyal girlfriend, and best girlfriend ever so...";

  stage = 1;

};


letter.onclick = ()=>{

  if(stage===1){

    letter.innerText='WILL YOU BE MY VALENTINE?';

    buttons.innerHTML='<button id="yes">Yes</button><button id="no">No</button>';

    stage=2;

    document.getElementById('yes').onclick=yesFlow;

    document.getElementById('no').onclick=noFlow;

  }

};


function yesFlow(){

  yesCount++;

  if(yesCount===1) message.innerText='ARE YOU SURE?';

  else if(yesCount<5) message.innerText='You are not sure enough you have to tap harder';

  else{

    cat.style.display='block';

    cat.innerText='🐱💐';

    message.innerText="Next year it's your turn";

    buttons.innerHTML='';

    confettiHearts();

  }

}


function noFlow(){

  noCount++;

  const texts=['why?','for real?','wow','SO YOU DON’T WANNA MAKE ME THE HAPPIEST OWAN?'];

  if(noCount<=texts.length){ message.innerText=texts[noCount-1]; }

  else{

    rainSound.play();

    cat.style.display='block';

    cat.innerText='😿';

    finger.style.display='block';

    message.innerText='You horrible person 😡 How could you?';

    tearRain();

  }

}


function confettiHearts(){

  setInterval(()=>{

    const heart = document.createElement('div');

    heart.className='heart';

    heart.innerText='💖';

    heart.style.left=Math.random()*100+'vw';

    document.body.appendChild(heart);

    setTimeout(()=>heart.remove(),4000);

  },200);

}


function tearRain(){

  setInterval(()=>{

    const tear = document.createElement('div');

    tear.className='tear';

    tear.innerText='💧';

    tear.style.left=Math.random()*100+'vw';

    document.body.appendChild(tear);

    setTimeout(()=>tear.remove(),4000);

  },300);

}


// Reset button

document.getElementById('reset').onclick=()=>{

  location.reload();

};
